cmake_minimum_required(VERSION 3.20.0)
option(JAVELIN_PROBE "Run the native Javelin dictionary reader from external flash" OFF)
if(JAVELIN_PROBE)
  list(APPEND EXTRA_CONF_FILE "${CMAKE_CURRENT_SOURCE_DIR}/javelin.conf")
endif()
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(zen_r4_qspi_probe)
target_sources(app PRIVATE src/main.c)

option(QUAD_PROBE "Build the separate quad-read diagnostic image" OFF)
option(SCRATCH_PROBE "Build the scratch-sector program and erase diagnostic image" OFF)
option(TRACE_PROBE "Repeat quad reads for waveform capture and timing comparison" OFF)
option(XIP_PROBE "Compare mapped quad reads with EasyDMA" OFF)
if(TRACE_PROBE AND NOT SCRATCH_PROBE)
  message(FATAL_ERROR "TRACE_PROBE requires SCRATCH_PROBE")
endif()
if(XIP_PROBE AND (NOT SCRATCH_PROBE OR TRACE_PROBE))
  message(FATAL_ERROR "XIP_PROBE requires SCRATCH_PROBE and a separate build from TRACE_PROBE")
endif()
if(JAVELIN_PROBE AND (NOT SCRATCH_PROBE OR XIP_PROBE OR TRACE_PROBE))
  message(FATAL_ERROR "JAVELIN_PROBE requires SCRATCH_PROBE and its own build directory")
endif()
if(QUAD_PROBE OR SCRATCH_PROBE)
  target_sources(app PRIVATE src/quad.c)
  target_compile_definitions(app PRIVATE QUAD_PROBE=1)
endif()
if(SCRATCH_PROBE)
  target_sources(app PRIVATE src/scratch.c)
  target_compile_definitions(app PRIVATE SCRATCH_PROBE=1)
endif()
if(TRACE_PROBE)
  target_sources(app PRIVATE src/trace.c)
  target_compile_definitions(app PRIVATE TRACE_PROBE=1)
endif()
if(XIP_PROBE OR JAVELIN_PROBE)
  target_sources(app PRIVATE src/mapped.c)
endif()
if(XIP_PROBE)
  target_compile_definitions(app PRIVATE XIP_PROBE=1)
endif()
if(JAVELIN_PROBE)
  include(javelin.cmake)
endif()
